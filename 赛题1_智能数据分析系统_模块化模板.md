### 2.1 核心数据对象

```python
# ============================================================
# 文件: backend/shared/schemas.py
# 作用: 全系统统一的数据结构定义，所有模块导入此文件
# ============================================================

from pydantic import BaseModel, Field
from typing import List, Dict, Optional, Any, Literal
from enum import Enum
from datetime import datetime

# ---------- 数据源相关 ----------

class DataSourceType(str, Enum):
    SQLITE = "sqlite"
    MYSQL = "mysql"
    POSTGRESQL = "postgresql"
    EXCEL = "excel"
    CSV = "csv"

class ColumnInfo(BaseModel):
    """字段元信息"""
    name: str                              # 字段名
    data_type: str                         # 数据类型: INTEGER/TEXT/REAL/DATE
    nullable: bool = True
    is_primary_key: bool = False
    is_foreign_key: bool = False
    referenced_table: Optional[str] = None # 外键引用的表
    sample_values: List[Any] = []          # 前5个样本值

class TableSchema(BaseModel):
    """表结构"""
    table_name: str
    columns: List[ColumnInfo]
    row_count: int
    description: str = ""                  # LLM生成的表描述

class DataSourceConfig(BaseModel):
    """数据源连接配置"""
    source_type: DataSourceType
    source_id: str                         # 唯一标识
    display_name: str                      # 界面显示名
    connection_params: dict                # {"file_path":"..."} 或 {"host":"...","port":...}

# ---------- 查询相关 ----------

class QueryIntent(str, Enum):
    SIMPLE_QUERY = "simple_query"          # 简单查询
    AGGREGATION = "aggregation"            # 聚合统计
    MULTI_TABLE = "multi_table"            # 多表关联
    FORMULA_CALC = "formula_calc"          # 公式计算
    ANOMALY_DETECT = "anomaly_detect"      # 异常检测
    FORECAST = "forecast"                  # 时序预测
    REPORT = "report"                      # 报告生成
    OTHER = "other"

class GeneratedQuery(BaseModel):
    """NL2SQL生成的查询对象"""
    query_type: Literal["sql", "python", "hybrid"]
    code: str                             # SQL语句 或 Python代码
    target_source_id: str                 # 目标数据源ID
    explanation: str                      # 对用户问题的理解转述
    clarifications: List[str] = []        # 需要用户澄清的问题

class QueryResult(BaseModel):
    """查询执行结果"""
    success: bool
    columns: List[str]                    # 结果列名
    data: List[List[Any]]                 # 结果数据(行)
    row_count: int
    execution_time_ms: float
    error_message: str = ""

# ---------- 对话相关 ----------

class Message(BaseModel):
    """单条对话消息"""
    role: Literal["user", "assistant", "system"]
    content: str
    timestamp: datetime
    attachments: List[Dict] = []           # [{type:"query_result", data: QueryResult}]

class Conversation(BaseModel):
    """一次完整对话"""
    conversation_id: str
    messages: List[Message]
    current_data_source_id: Optional[str] = None
    created_at: datetime

# ---------- 可视化相关 ----------

class ChartType(str, Enum):
    LINE = "line"
    BAR = "bar"
    PIE = "pie"
    SCATTER = "scatter"
    HEATMAP = "heatmap"

class ChartConfig(BaseModel):
    """图表配置"""
    chart_type: ChartType
    title: str
    x_axis: str                           # X轴字段名
    y_axis: str                           # Y轴字段名
    series_data: Dict[str, List]          # {legend_name: [values]}
    options: dict = {}                    # ECharts option 覆盖项

class AnalysisReport(BaseModel):
    """分析报告"""
    title: str
    summary: str                          # 分析概述
    key_metrics: List[Dict]               # [{"label":"总销售额","value":"1.2亿"}]
    charts: List[ChartConfig]
    insights: List[str]                   # 关键发现
    recommendations: List[str]            # 建议

# ---------- API请求/响应 ----------

class ChatRequest(BaseModel):
    """对话请求"""
    conversation_id: str
    question: str
    data_source_ids: List[str] = []       # 本次查询使用的数据源

class ChatResponse(BaseModel):
    """对话响应"""
    conversation_id: str
    answer_text: str                      # 自然语言回答
    query_result: Optional[QueryResult] = None
    chart: Optional[ChartConfig] = None
    report: Optional[AnalysisReport] = None
    suggested_questions: List[str] = []   # 建议追问
```

```tsx
export const ChatPanel: React.FC<ChatPanelProps> = ({ conversationId, onNewConversation }) => {

    <div className="chat-panel">
      <MessageList messages={messages} loading={loading} />
      <InputArea value={input} onChange={setInput} onSend={sendMessage} />
      <SuggestedQuestions onClick={sendMessage} />

  );
};
```

**Streamlit 实现骨架** (推荐快速原型):

```python
# frontend/app.py
import streamlit as st
import requests

st.set_page_config(page_title="智能数据分析助手", layout="wide")

# 初始化会话状态
if "messages" not in st.session_state:
    st.session_state.messages = []
if "conversation_id" not in st.session_state:
    st.session_state.conversation_id = str(uuid.uuid4())

# 对话区
for msg in st.session_state.messages:
    with st.chat_message(msg["role"]):
        st.markdown(msg["content"])

# 输入框
if question := st.chat_input("请输入您的问题..."):
    st.session_state.messages.append({"role": "user", "content": question})
    with st.chat_message("assistant"):
        with st.spinner("正在分析..."):
            resp = requests.post("http://localhost:8000/api/chat", json={
                "conversation_id": st.session_state.conversation_id,
                "question": question,
                "data_source_ids": st.session_state.get("active_sources", [])
            })
            data = resp.json()
            st.markdown(data["answer_text"])
            if data.get("chart"):
                render_chart(data["chart"])  # 见模块5
    st.session_state.messages.append({"role": "assistant", "content": data["answer_text"]})
```

```python
# backend/nlp/intent_classifier.py
from shared.schemas import QueryIntent

class IntentClassifier:
    """
    输入: 用户问题(str) + 对话历史(List[Message])
    输出: QueryIntent 枚举值
    
    分类规则 + LLM兜底:
    1. 关键词匹配(快速路径): 
       "统计/汇总/平均/总共" → AGGREGATION
       "关联/联合/对应"       → MULTI_TABLE
       "预测/趋势/未来"       → FORECAST
       "异常/离群/不正常"     → ANOMALY_DETECT
       "报告/总结/分析报告"   → REPORT
    2. 规则无法匹配时 → 调用LLM分类
    """
    
    KEYWORD_MAP = {
        QueryIntent.AGGREGATION:  ["统计","汇总","求和","平均","最多","最少","排名"],
        QueryIntent.MULTI_TABLE:  ["关联","联合","对应","结合","跨表"],
        QueryIntent.FORMULA_CALC: ["计算","同比","环比","占比","比率"],
        QueryIntent.FORECAST:     ["预测","趋势","走势","未来","预估"],
        QueryIntent.ANOMALY_DETECT: ["异常","离群","突出","不正常"],
        QueryIntent.REPORT:       ["报告","总结","综合分析"],
    }
    
    def classify(self, question: str, history: list = None) -> QueryIntent:
        # 1. 关键词快速匹配
        for intent, keywords in self.KEYWORD_MAP.items():
            if any(kw in question for kw in keywords):
                return intent
        # 2. LLM兜底
        return self._llm_classify(question)
    
    def _llm_classify(self, question: str) -> QueryIntent:
        prompt = f"""判断用户问题的意图类型，选项:
        - simple_query: 简单查询和筛选
        - aggregation: 聚合统计
        - multi_table: 多表关联
        - formula_calc: 公式/比率计算
        - anomaly_detect: 异常检测
        - forecast: 预测
        - report: 生成报告
        - other: 其他
        
        用户问题: {question}
        意图:"""
        result = call_llm(prompt)
        return QueryIntent(result.strip())
```

### 子模块2-2：Schema映射器 (schema_mapper)

```
文件: backend/nlp/schema_mapper.py

职责: 获取数据源的表结构信息，格式化为LLM可理解的描述
```

```python
# backend/nlp/schema_mapper.py
from shared.schemas import TableSchema, ColumnInfo, DataSourceConfig
from typing import List

class SchemaMapper:
    """
    输入: DataSourceConfig (通过模块3的适配器获取实际Schema)
    输出: 格式化后的Schema文本, 供Prompt使用
    
    关键功能:
    - 字段名中英文映射提示
    - 关联关系标注
    - 数值/日期/文本字段分类
    """
    
    def format_for_prompt(self, schemas: List[TableSchema]) -> str:
        """将多表Schema格式化为Prompt可用的文本"""
        parts = []
        for table in schemas:
            lines = [f"\n## 表名: {table.table_name}"]
            if table.description:
                lines.append(f"描述: {table.description}")
            lines.append(f"行数: {table.row_count}")
            lines.append("字段:")
            for col in table.columns:
                pk_flag = " [主键]" if col.is_primary_key else ""
                fk_flag = f" [外键→{col.referenced_table}]" if col.is_foreign_key else ""
                sample = f" 示例: {col.sample_values[:3]}" if col.sample_values else ""
                lines.append(f"  - {col.name} ({col.data_type}){pk_flag}{fk_flag}{sample}")
            parts.append("\n".join(lines))
        return "\n".join(parts)
    
    def suggest_joins(self, schemas: List[TableSchema]) -> List[dict]:
        """自动发现表间可能的关联关系"""
        joins = []
        for i, t1 in enumerate(schemas):
            for j, t2 in enumerate(schemas):
                if i >= j: continue
                for c1 in t1.columns:
                    for c2 in t2.columns:
                        # 字段名相同 → 可能关联
                        if c1.name.lower() == c2.name.lower():
                            joins.append({
                                "table1": t1.table_name,
                                "table2": t2.table_name,
                                "on": c1.name,
                                "confidence": "medium"
                            })
                        # 命名规律: xxx_id → 表名
                        if c1.name.lower() == f"{t2.table_name.lower()}_id":
                            joins.append({
                                "table1": t1.table_name,
                                "table2": t2.table_name,
                                "on": c1.name,
```

```python
# backend/nlp/sql_validator.py
import sqlparse
import re

class SQLValidator:
    """
    输入: GeneratedQuery.code (SQL字符串)
    输出: (is_valid: bool, errors: List[str], sanitized_sql: str)
    
    校验项:
    1. 语法检查: 用sqlparse解析，判断是否为有效SQL
    2. 安全检查: 禁止 DROP/INSERT/UPDATE/DELETE/ALTER/TRUNCATE
    3. 表名/字段名白名单: 只允许Schema中存在的表和字段
    4. 长度限制: 单条SQL不超过10000字符
    """
    
    FORBIDDEN_KEYWORDS = [
        "DROP", "INSERT", "UPDATE", "DELETE", "ALTER", 
        "TRUNCATE", "CREATE", "EXEC", "EXECUTE", "--"
    ]
    
    def validate(self, sql: str, allowed_tables: List[str], allowed_columns: List[str]) -> tuple:
        errors = []
        
        # 1. 安全检查
        sql_upper = sql.upper()
        for kw in self.FORBIDDEN_KEYWORDS:
            if kw in sql_upper:
                errors.append(f"禁止的操作: {kw}")
        
        # 2. 语法检查
        try:
            parsed = sqlparse.parse(sql)
            if not parsed:
                errors.append("SQL语法无效")
        except:
            errors.append("SQL解析失败")
        
        # 3. 长度检查
        if len(sql) > 10000:
            errors.append("SQL语句过长")
        
        return (len(errors) == 0, errors, sql)
```

```python
from abc import ABC, abstractmethod
from typing import List
from shared.schemas import TableSchema, QueryResult, ColumnInfo, DataSourceConfig
import pandas as pd

class BaseAdapter(ABC):
    """所有数据源适配器的基类"""
    
    def __init__(self, config: DataSourceConfig):
        self.config = config
        self._connected = False
    
    @abstractmethod
    def connect(self) -> bool:
        """建立连接，返回是否成功"""
        ...
    
    @abstractmethod
    def get_tables(self) -> List[str]:
        """获取所有表/Sheet名称"""
        ...
    
    @abstractmethod
    def get_schema(self, table_name: str) -> TableSchema:
        """获取单表结构"""
        ...
    
    @abstractmethod
    def execute(self, sql: str) -> QueryResult:
        """执行SQL查询"""
        ...
    
    def get_all_schemas(self) -> List[TableSchema]:
        """获取所有表的Schema(批量)"""
        return [self.get_schema(t) for t in self.get_tables()]
    
    def preview(self, table_name: str, n: int = 5) -> pd.DataFrame:
        """预览前N行数据"""
        result = self.execute(f"SELECT * FROM {table_name} LIMIT {n}")
        return pd.DataFrame(result.data, columns=result.columns)
    
    def disconnect(self):
        """关闭连接"""
        self._connected = False
```

```python
import sqlite3
import os
from shared.schemas import QueryResult, TableSchema, ColumnInfo
from .base import BaseAdapter

class SQLiteAdapter(BaseAdapter):
    def connect(self) -> bool:
        path = self.config.connection_params["file_path"]
        if not os.path.exists(path):
            raise FileNotFoundError(f"SQLite文件不存在: {path}")
        self._conn = sqlite3.connect(path)
        self._conn.row_factory = sqlite3.Row
        self._connected = True
        return True
    
    def get_tables(self) -> List[str]:
        cursor = self._conn.execute(
            "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'"
        )
        return [row[0] for row in cursor.fetchall()]
    
    def get_schema(self, table_name: str) -> TableSchema:
        cursor = self._conn.execute(f"PRAGMA table_info({table_name})")
        columns = []
        for row in cursor.fetchall():
            columns.append(ColumnInfo(
                name=row[1],
                data_type=row[2],
                nullable=not row[3],
                is_primary_key=bool(row[5]),
            ))
        
        count = self._conn.execute(f"SELECT COUNT(*) FROM {table_name}").fetchone()[0]
        return TableSchema(table_name=table_name, columns=columns, row_count=count)
    
    def execute(self, sql: str) -> QueryResult:
        import time
        start = time.time()
        try:
            cursor = self._conn.execute(sql)
            columns = [desc[0] for desc in cursor.description] if cursor.description else []
            data = [list(row) for row in cursor.fetchall()]
            return QueryResult(
                success=True,
                columns=columns,
                data=data,
                row_count=len(data),
                execution_time_ms=(time.time()-start)*1000,
            )
        except Exception as e:
            return QueryResult(success=False, columns=[], data=[], 
                             row_count=0, execution_time_ms=0, error_message=str(e))
```

```python
import pandas as pd

class SQLAlchemyAdapter(BaseAdapter):
    """MySQL/PostgreSQL适配器共用逻辑"""
    
    def connect(self) -> bool:
        db_type = "mysql+pymysql" if self.config.source_type == "mysql" else "postgresql+psycopg2"
        params = self.config.connection_params
        url = f"{db_type}://{params['user']}:{params['password']}@{params['host']}:{params['port']}/{params['database']}"
        self._engine = create_engine(url)
        self._inspector = inspect(self._engine)
        self._connected = True
        return True
    
    def get_tables(self) -> List[str]:
        return self._inspector.get_table_names()
    
    def get_schema(self, table_name: str) -> TableSchema:
        cols = self._inspector.get_columns(table_name)
        pk = self._inspector.get_pk_constraint(table_name)
        fks = self._inspector.get_foreign_keys(table_name)
        pk_cols = set(pk.get("constrained_columns", []))
        fk_map = {fk["constrained_columns"][0]: fk["referred_table"] for fk in fks if fk["constrained_columns"]}
        
        columns = []
        for col in cols:
            columns.append(ColumnInfo(
                name=col["name"],
                data_type=str(col["type"]),
                nullable=col.get("nullable", True),
                is_primary_key=col["name"] in pk_cols,
                is_foreign_key=col["name"] in fk_map,
                referenced_table=fk_map.get(col["name"]),
            ))
        
        row_count = pd.read_sql(f"SELECT COUNT(*) FROM {table_name}", self._engine).iloc[0,0]
        return TableSchema(table_name=table_name, columns=columns, row_count=int(row_count))
    
    def execute(self, sql: str) -> QueryResult:
        import time; start = time.time()
        try:
            df = pd.read_sql(sql, self._engine)
            return QueryResult(success=True, columns=list(df.columns),
                             data=df.values.tolist(), row_count=len(df),
                             execution_time_ms=(time.time()-start)*1000)
```

```python
    1. 外键约束 (最可靠)
    2. 字段名匹配 (如user_id, dept_id)
    3. 值域重叠度 (两个字段的值域高度重叠)
    """
    
    def discover_joins(self, schemas: List[TableSchema]) -> List[dict]:
        """自动发现表关联"""
        joins = []
        for i, t1 in enumerate(schemas):
            for j, t2 in enumerate(schemas):
                if i >= j: continue
                # 检查外键
                for c1 in t1.columns:
                    for c2 in t2.columns:
                        if c1.is_foreign_key and c1.referenced_table == t2.table_name:
                            joins.append({"left": t1.table_name, "right": t2.table_name,
                                         "on": c1.name, "type": "fk", "confidence": 1.0})
                        # 字段名相同
                        if c1.name.lower() == c2.name.lower() and c1.data_type == c2.data_type:
                            joins.append({"left": t1.table_name, "right": t2.table_name,
                                         "on": c1.name, "type": "name_match", "confidence": 0.7})
        return joins
```

```python
# backend/analyzer/anomaly_detector.py
from sklearn.ensemble import IsolationForest
from sklearn.preprocessing import StandardScaler
import pandas as pd
import numpy as np

class AnomalyDetector:
    """
    输入: QueryResult (DataFrame)
    输出: 异常数据行 + 异常分数
    
    三种方法:
    - Z-Score: 基于标准差，适合单变量
    - IQR: 基于四分位距，鲁棒性好
    - Isolation Forest: 适合多变量异常检测
    """
    
    def zscore_detect(self, df: pd.DataFrame, column: str, threshold: float = 3.0) -> pd.DataFrame:
        """Z-Score异常检测"""
        z = np.abs((df[column] - df[column].mean()) / df[column].std())
        df["is_anomaly"] = z > threshold
        df["anomaly_score"] = z
        return df[df["is_anomaly"]]
    
    def iqr_detect(self, df: pd.DataFrame, column: str) -> pd.DataFrame:
        """IQR异常检测"""
        Q1, Q3 = df[column].quantile(0.25), df[column].quantile(0.75)
        IQR = Q3 - Q1
        lower, upper = Q1 - 1.5*IQR, Q3 + 1.5*IQR
        df["is_anomaly"] = (df[column] < lower) | (df[column] > upper)
        return df[df["is_anomaly"]]
    
    def isolation_forest(self, df: pd.DataFrame, numeric_cols: list) -> pd.DataFrame:
        """Isolation Forest多变量异常检测"""
        X = df[numeric_cols].select_dtypes(include=[np.number]).fillna(0)
        model = IsolationForest(contamination=0.05, random_state=42)
        df["is_anomaly"] = model.fit_predict(X) == -1
        df["anomaly_score"] = model.decision_function(X)
        return df[df["is_anomaly"]]
```

```python
from prophet import Prophet
import pandas as pd

class Forecaster:
    """
    输入: QueryResult (DataFrame, 需含日期列+数值列)
    输出: 预测结果DataFrame (含置信区间)
    
    支持方法:
    - Prophet (推荐，对业务友好)
    - ARIMA (备选)
    """
    
    def forecast_prophet(self, df: pd.DataFrame, date_col: str, value_col: str, 
                         periods: int = 30) -> pd.DataFrame:
        """使用Prophet进行时序预测"""
        prophet_df = df[[date_col, value_col]].copy()
        prophet_df.columns = ["ds", "y"]
        prophet_df["ds"] = pd.to_datetime(prophet_df["ds"])
        
        model = Prophet(
            yearly_seasonality=True,
            weekly_seasonality=True,
            daily_seasonality=False,
        )
        model.fit(prophet_df)
        future = model.make_future_dataframe(periods=periods)
        forecast = model.predict(future)
        return forecast[["ds", "yhat", "yhat_lower", "yhat_upper"]]
```

```python
# backend/visualizer/chart_recommender.py
from shared.schemas import ChartType, ChartConfig, QueryResult
import pandas as pd

class ChartRecommender:
    """
    输入: QueryResult
    输出: ChartConfig (推荐图表类型 + 轴映射)
    
    推荐规则表:
    | 列类型组合              | 推荐图表  | 理由                  |
    |------------------------|----------|----------------------|
    | 1日期 + 1数值           | LINE     | 时间趋势              |
    | 1类别(≤10) + 1数值     | BAR      | 对比排名              |
    | 1类别 + 1数值(占比)     | PIE      | 构成占比              |
    | 2数值                  | SCATTER  | 相关性                |
    | 1类别 + 2+数值         | BAR      | 分组对比              |
    | 矩阵(行列+数值)         | HEATMAP  | 热力分布              |
    """
    
    def recommend(self, result: QueryResult) -> ChartConfig:
        df = pd.DataFrame(result.data, columns=result.columns)
        cols = result.columns
        
        date_cols = self._find_date_columns(df)
        num_cols = self._find_numeric_columns(df)
        cat_cols = [c for c in cols if c not in date_cols and c not in num_cols]
        
        # 规则1: 日期 + 数值 → 折线图
        if date_cols and num_cols:
            return ChartConfig(
                chart_type=ChartType.LINE,
                title="",
                x_axis=date_cols[0],
                y_axis=num_cols[0],
                series_data={num_cols[0]: df[num_cols[0]].tolist()},
            )
        # 规则2: 类别(≤10类) + 数值 → 柱状图
        if cat_cols and num_cols:
            unique = df[cat_cols[0]].nunique()
            if 2 <= unique <= 15:
                return ChartConfig(
                    chart_type=ChartType.PIE if unique <= 6 else ChartType.BAR,
                    title="",
                    x_axis=cat_cols[0],
                    y_axis=num_cols[0],
                    series_data={num_cols[0]: df[num_cols[0]].tolist()},
                )
        # 规则3: 2个数值列 → 散点图
        if len(num_cols) >= 2:
            return ChartConfig(
                chart_type=ChartType.SCATTER,
                title="", x_axis=num_cols[0], y_axis=num_cols[1],
                series_data={"data": df[[num_cols[0], num_cols[1]]].values.tolist()},
            )
        
        # 默认: 柱状图
        first_num = num_cols[0] if num_cols else cols[-1]
        first_label = cols[0] if cols[0] != first_num else (cols[1] if len(cols)>1 else cols[0])
        return ChartConfig(chart_type=ChartType.BAR, title="", x_axis=first_label, y_axis=first_num, series_data={})
    
    def _find_date_columns(self, df: pd.DataFrame) -> list:
        return [c for c in df.columns if pd.api.types.is_datetime64_any_dtype(df[c])]
    
    def _find_numeric_columns(self, df: pd.DataFrame) -> list:
        return [c for c in df.columns if pd.api.types.is_numeric_dtype(df[c])]
```

```python
class EChartsBuilder:
    """将ChartConfig转化为ECharts的option字典"""
    
    def build(self, config: ChartConfig, result_data: list, columns: list) -> dict:
        builders = {
            ChartType.LINE: self._build_line,
            ChartType.BAR: self._build_bar,
            ChartType.PIE: self._build_pie,
            ChartType.SCATTER: self._build_scatter,

            "series": [{
                "name": cfg.y_axis, "type": "line",
                "data": [row[1] for row in data],
                "smooth": True,
            }],

            "yAxis": {"type": "value"},
            "series": [{
                "name": cfg.y_axis, "type": "bar",
                "data": [row[1] for row in data],
                "itemStyle": {"borderRadius": [4,4,0,0]},

            "title": {"text": cfg.title or "数据对比"},
```

```python
class ReportGenerator:
    """

    输出: AnalysisReport (结构化报告)

用户问题: {question}
分析过程: {analysis_steps}
查询结果: {result_summary}

请输出一份结构化报告(JSON格式):
{{
  "title": "报告标题",
  "summary": "一段话概述核心发现(100字内)",
  "key_metrics": [{{"label":"指标名","value":"数值","change":"+5%","trend":"up"}}],
  "insights": ["发现1", "发现2", "发现3"],
  "recommendations": ["建议1", "建议2"]
}}
"""
    
    def generate(self, question: str, results: List[QueryResult], 
                 analysis_steps: str = "") -> AnalysisReport:
        result_summary = "\n".join([
            f"查询{r+1}: {res.row_count}行, 列: {res.columns}"
            for r, res in enumerate(results) if res.success
        ])
        prompt = self.REPORT_PROMPT.format(
            question=question,
            analysis_steps=analysis_steps,
            result_summary=result_summary,
        )
        llm_output = call_llm(prompt)
        return self._parse_report(llm_output)
    
    def _parse_report(self, raw: str) -> AnalysisReport:
        import json, re
        match = re.search(r'\{.*\}', raw, re.DOTALL)
        data = json.loads(match.group())
        return AnalysisReport(**data)
```

```python
import logging, json, time

class AnalysisLogger:
    """记录每次分析的完整链路"""
    
    def log_query(self, conversation_id, question, intent, generated_sql, 
                  validation_result, execution_time, row_count, success):
        log_entry = {
            "timestamp": time.strftime("%Y-%m-%d %H:%M:%S"),
            "conversation_id": conversation_id,
            "question": question,
            "intent": intent,
            "generated_sql": generated_sql,
            "sql_valid": validation_result,
            "execution_time_ms": execution_time,
            "row_count": row_count,
            "success": success,
        }
        logging.info(json.dumps(log_entry, ensure_ascii=False))
```

### 关键技术决策

| 决策 | 选择 | 原因 |
|------|------|------|
| LLM框架 | LangChain + ChatOpenAI | 组件化Prompt，预编译Chain，切换Provider只需改base_url |
| JSON解析 | StrOutputParser + 手动三重解析 | DeepSeek对JsonOutputParser不稳定，手动解析更可靠 |
| 密钥管理 | Windows注册表读取 | setx后无需重启终端即可生效，密钥不落盘 |
| 文件数据源 | DuckDB | 让CSV/Excel支持SQL查询，无需导入数据库 |
| 关系型数据库 | SQLAlchemy | MySQL/PostgreSQL共用一个基类，减少重复代码 |
| 图表构建 | ChartRecommender → EChartsBuilder | 推荐与构建分离，易于扩展新图表类型 |
| 表名安全 | 数字开头自动加t_前缀 | SQL标识符不能以数字开头 |
